cmake_minimum_required(VERSION 3.20)
project(OpenSSHManager VERSION 0.1.23 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_library(manager_config src/config.cpp src/diagnostics.cpp src/authorized_keys.cpp src/install_marker.cpp src/i18n.cpp)
target_include_directories(manager_config PUBLIC src)
if(MSVC)
  target_compile_options(manager_config PRIVATE /W4 /utf-8)
else()
  target_compile_options(manager_config PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  if(NOT CMAKE_CROSSCOMPILING)
    find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell)
    if(POWERSHELL_EXECUTABLE)
      foreach(script localization install_mode install_data engine_path engine_archive)
        add_test(NAME ${script}_powershell
          COMMAND "${POWERSHELL_EXECUTABLE}" -NoLogo -NoProfile -ExecutionPolicy Bypass
            -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/${script}_tests.ps1")
      endforeach()
    endif()
  endif()
  add_executable(i18n_tests tests/i18n_tests.cpp)
  target_link_libraries(i18n_tests PRIVATE manager_config)
  if(MSVC)
    target_compile_options(i18n_tests PRIVATE /utf-8)
  endif()
  add_test(NAME language_selection_and_catalog COMMAND i18n_tests)
  add_executable(config_tests tests/config_tests.cpp)
  target_link_libraries(config_tests PRIVATE manager_config)
  if(WIN32)
    target_link_libraries(config_tests PRIVATE shell32)
  endif()
  add_test(NAME config_transactions COMMAND config_tests)
  add_executable(diagnostics_tests tests/diagnostics_tests.cpp)
  target_link_libraries(diagnostics_tests PRIVATE manager_config)
  add_test(NAME login_diagnostics COMMAND diagnostics_tests)
  add_executable(scroll_tests tests/scroll_tests.cpp)
  target_include_directories(scroll_tests PRIVATE src)
  add_test(NAME capsule_scroll_geometry COMMAND scroll_tests)
  add_executable(setup_action_tests tests/setup_action_tests.cpp)
  target_include_directories(setup_action_tests PRIVATE src)
  add_test(NAME staged_setup_actions COMMAND setup_action_tests)
  add_executable(layout_tests tests/layout_tests.cpp)
  target_include_directories(layout_tests PRIVATE src)
  add_test(NAME proportional_layout COMMAND layout_tests)
  add_executable(authorized_keys_tests tests/authorized_keys_tests.cpp)
  target_link_libraries(authorized_keys_tests PRIVATE manager_config)
  add_test(NAME authorized_key_entries COMMAND authorized_keys_tests)
  if(MSVC)
    target_compile_options(config_tests PRIVATE /utf-8)
    target_compile_options(diagnostics_tests PRIVATE /utf-8)
  endif()
endif()
if(WIN32)
  enable_language(RC)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/engine.lock.json" ENGINE_LOCK)
  string(JSON ENGINE_URL GET "${ENGINE_LOCK}" url)
  string(JSON ENGINE_SHA256 GET "${ENGINE_LOCK}" sha256)
  string(JSON ENGINE_TAG GET "${ENGINE_LOCK}" tag)
  configure_file(cmake/engine_release.h.in generated/engine_release.h @ONLY)
  configure_file(cmake/setup_resources.rc.in generated/setup_resources.rc @ONLY)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS engine.lock.json)
  set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/generated/setup_resources.rc" PROPERTY OBJECT_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/scripts/Install.ps1;${CMAKE_CURRENT_SOURCE_DIR}/scripts/EnginePath.ps1;${CMAKE_CURRENT_SOURCE_DIR}/scripts/EngineArchive.ps1;${CMAKE_CURRENT_SOURCE_DIR}/scripts/Uninstall.ps1;${CMAKE_CURRENT_SOURCE_DIR}/engine.lock.json;${CMAKE_CURRENT_SOURCE_DIR}/scripts/Localization.ps1")
  add_custom_target(manager_build_info
    COMMAND "${CMAKE_COMMAND}" "-DVERSION=${PROJECT_VERSION}" "-DOUTPUT_DIR=${CMAKE_CURRENT_BINARY_DIR}/generated"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/build_info.cmake"
    BYPRODUCTS "${CMAKE_CURRENT_BINARY_DIR}/generated/build_info.h"
    VERBATIM)

  add_executable(OpenSSHManager WIN32 src/main.cpp src/platform.cpp src/bootstrap.cpp src/system_info.cpp src/events.cpp src/theme.cpp src/theme_controls.cpp src/scrollbar.cpp src/key_channel.cpp src/key_popup.cpp resources/app.rc "${CMAKE_CURRENT_BINARY_DIR}/generated/setup_resources.rc")
  add_dependencies(OpenSSHManager manager_build_info)
  target_include_directories(OpenSSHManager PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(OpenSSHManager PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(OpenSSHManager PRIVATE manager_config comctl32 comdlg32 shell32 advapi32 ole32 oleaut32 uuid bcrypt wevtapi ws2_32 userenv netapi32 gdiplus uxtheme dwmapi version iphlpapi winhttp)
  if(MSVC)
    target_compile_options(OpenSSHManager PRIVATE /W4 /utf-8)
    target_link_options(OpenSSHManager PRIVATE /MANIFEST:NO)
  else()
    target_compile_options(OpenSSHManager PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(OpenSSHManager PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  if(BUILD_TESTING)
    add_executable(native_controls_tests tests/native_controls_tests.cpp src/platform.cpp src/bootstrap.cpp src/system_info.cpp src/theme.cpp src/theme_controls.cpp src/scrollbar.cpp src/key_channel.cpp src/key_popup.cpp resources/app.rc "${CMAKE_CURRENT_BINARY_DIR}/generated/setup_resources.rc")
    target_include_directories(native_controls_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(native_controls_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_link_libraries(native_controls_tests PRIVATE manager_config comctl32 comdlg32 shell32 advapi32 ole32 oleaut32 uuid bcrypt wevtapi ws2_32 userenv netapi32 gdiplus uxtheme dwmapi version iphlpapi winhttp)
    if(MSVC)
      target_compile_options(native_controls_tests PRIVATE /W4 /utf-8)
      target_link_options(native_controls_tests PRIVATE /MANIFEST:NO)
    else()
      target_compile_options(native_controls_tests PRIVATE -Wall -Wextra -Wpedantic)
      target_link_options(native_controls_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME native_controls COMMAND native_controls_tests)
    set_tests_properties(native_controls PROPERTIES TIMEOUT 60)
  endif()
  install(TARGETS OpenSSHManager RUNTIME DESTINATION .)
endif()
